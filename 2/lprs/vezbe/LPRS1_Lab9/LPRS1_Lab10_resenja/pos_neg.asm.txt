
/*
	Ekvivalentan C kod:

	int N = 10;
	short a[10] = {-3, 0, 0, 7, 12, -6, 15, -1, 0, 11};
*/
.data
10	// N = 10
-3	// a[0], adresa 0
0 	// a[1], adresa 1
0 	// a[2], adresa 2
7 	// a[3], adresa 3
12 	// a[4], adresa 4
-6 	// a[5], adresa 5
15 	// a[6], adresa 6
-1 	// a[7], adresa 7
0 	// a[8], adresa 8
11 	// a[9], adresa 9

/*
	Ekvivalentan C kod (viši nivo):
	
	for(i=0; i<N; i++)
	{
		R1 = a[i];
		if(R1 == 0)
			R5++;
		elsif(R1 < 0)
			R6++;
		else
			R7++;
	}
*/
.text
/*
	Spisak registara:

	R0 - i - adresa elementa iz memorije.
	R1 - učitan element iz memorije a[i]
	R3 - N = 10
	R5 - broj NULA
	R6 - broj negativnih
	R7 - broj pozitivnih
*/
begin:
	ld R3, R0		// R3 = N = 10, prva vrednost u memoriji je broj elemenata
	sub R0, R0, R0	// R0 = R0 - R0, R0 = 0
for:
	sub R2, R3, R0	// for(i = 0, i<10, i++)
	jmpz end		// ako je i > N , kraj
	inc R0, R0		// i++, vrednost uvećavamo pre LD jer je a[0] = broj elemenata
	ld R1, R0		// R1 = a[i]
	jmpz NULA		// if a[i] = 0 -> NULA++
	jmps NEG		// if a[i] = 0 -> NEG++
POZ:
	inc R7, R7		// else POZ++
	jmp for
NULA:
	inc R5, R5		// NEG++
	jmp for
NEG:
	inc R6, R6		// NULA++
	jmp for
end:
	jmp end // Beskonačna petlja.
