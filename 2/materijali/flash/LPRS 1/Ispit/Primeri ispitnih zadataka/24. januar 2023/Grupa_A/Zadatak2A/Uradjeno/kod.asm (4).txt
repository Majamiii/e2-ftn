/*
	// .data
	
	short N = 10;				// N = 10
	short* p_result = 0x30;	// adresa novog niza
	short* p_done	 = 0x50;	// adresa za kraj programa

	short a[10] = {5, -3, 7, -6, -5, -4, -1, 0, 1, -7};
*/

.data
10
0x30
0x50
5, -3, 7, -6, -5, -4, -1, 0, 1, -7

/*
	// .text
	for(short i=0; i<N; i++)
	{
		if(a[i] < 0) and a[i]%2 != 0)	
          *p_result = a[i];
		p_result++;
	}

	*p_done = 1;
*/

.text
/*
	Recommended register list:
	R0 - N
	R1 - p_result
	R2 - p_done
	R3 - a[i]
	R4 - a[i] koji zadovoljava uslove
	R5 - i
	R6 - tmp
	R7 - tmp
*/

begin:
	ld R0, R0	// N = 10
	
	// niz krece od adrese 3, R6
	inc R6, R6
	inc R6, R6
	inc R6, R6
	inc R1, R1
	ld R1, R1	// R1 = 0x30 = *p_result
for:
	sub R7, R0, R5	// R7 = N - i
	jmpz end_for
	ld R3, R6	// R3 = a[i]
	jmpns end_if
	sub R7, R7, R7	
	inc R7, R7	// R7 = 1 maska
	and R7, R3, R7
	jmpz end_if
	st R3, R1	// rez = a[i]
	inc R1, R1	// p_result++
end_if:
	inc R5, R5	// i za for petlju ++
	inc R6, R6	// i za niz ++
	jmp for
end_for:
	inc R2, R2	// R2 = 1
	inc R2, R2
	ld R2, R2	// R2 = 0x50
	sub R0, R0, R0
	inc R0, R0
	st R0, R2
end:
	jmp end
