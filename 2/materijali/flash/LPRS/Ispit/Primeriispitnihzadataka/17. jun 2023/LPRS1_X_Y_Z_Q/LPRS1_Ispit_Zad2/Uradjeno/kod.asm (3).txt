/*
	// .data
	short* p_result	= 0x1B;
	short* p_done		= 0x20;
	short N = 5;
	short a[5] = {-1, -2, -3, -4,  -7};
*/
.data
0x1B
0x20
5
-1, -2, -3, -4, -7

/*
	// .text
	short res = 0;
	
	for(int i = 0; i<N; i++)
	{
		res += 2^i - a[i];
	}
	res /= 16;
	
	if(res < 1)
		*p_result = res;
	else
		*(p_result+1) = res;

	*p_done = 1;
*/

.text
/*
	Recommended register list:
	R0 - N
	R1 - i
	R2 - pow
	R3 - a[i]
	R4 - p_result
	R5 - p_done
	R6 - tmp
	R7 - res
*/
begin:
	inc R0, R0
	inc R0, R0
	ld R0, R0	// N = 5
	inc R3, R3
	inc R3, R3
	inc R3, R3
	mov R4, R3
	inc R2, R2	// R2 = 1; 2 na 0 je 1
for:
	mov R3, R4
	ld R3, R3	// a[i], 1.clan niza
	sub R6, R1, R0
	jmpz end_for
	add R7, R7, R2	// R7 = R7 + 2 na i
	sub R7, R7, R3	// R7 = R7 - a[i]
	ashl R2, R2	// mnozenje sa 2
	inc R1, R1
	inc R4, R4
	jmp for
end_for:
	// res = res / 16
	ashr R7, R7
	ashr R7, R7
	ashr R7, R7
	ashr R7, R7
if:
	sub R4, R4, R4
	sub R0, R0, R0
	inc R0, R0	// R0 = 1
	sub R7, R7, R0
	jmpns else
	ld R4, R4	// R4 = 0x1B
	st R7, R4	// R4 = res
	jmp end_if
else:
	ld R4, R4	// R4 = 0x1B
	inc R4, R4	
	st R7, R4	// R4 = res
end_if:
	sub R0, R0, R0
	inc R0, R0
	inc R5, R5
	ld R5, R5	// R5 = 0x20
	st R0, R5
end:
	jmp end
